Chapitre 6Supabase — Base de données

Supabase est le backend de SenMoto. C'est un BaaS (Backend as a Service) open-source qui fournit une base de données PostgreSQL, une API REST auto-générée, une authentification et un système de fichiers — le tout hébergé dans le cloud.

6.6 — Fonctions RPC

Quand une opération est trop complexe pour PostgREST (ou requiert des privilèges spéciaux), on crée une fonction SQL côté Supabase et on l'appelle depuis React Native. 

// Côté Supabase (SQL) — Exemple : créer entreprise + admin en 1 transaction
CREATE OR REPLACE FUNCTION create_enterprise_with_admin(
  p_name TEXT,
  p_code TEXT,
  p_phone TEXT,
  p_email TEXT,
  p_logo_url TEXT,
  p_admin_user_id UUID,
  p_admin_full_name TEXT,
  p_admin_email TEXT,
  p_admin_phone TEXT
) RETURNS JSON
LANGUAGE plpgsql
SECURITY DEFINER  -- Exécute avec les droits du créateur, pas de l'appelant
AS $$
DECLARE
  v_enterprise_id UUID;
BEGIN
  -- Créer l'entreprise
  INSERT INTO enterprises (name, code, phone, email, logo_url, created_by)
  VALUES (p_name, p_code, p_phone, p_email, p_logo_url, p_admin_user_id)
  RETURNING id INTO v_enterprise_id;

  -- Créer l'admin
  INSERT INTO enterprise_admins (enterprise_id, user_id, full_name, email, phone)
  VALUES (v_enterprise_id, p_admin_user_id, p_admin_full_name, p_admin_email, p_admin_phone);

  RETURN json_build_object('enterprise_id', v_enterprise_id, 'success', true);
END;
$$;

// Côté React Native — Appel RPC (lib/multitenant.ts)

📌 SECURITY DEFINER — Contourner RLS Normalement, un utilisateur ne peut pas créer d'entreprises pour lui-même (RLS bloque). SECURITY DEFINER exécute la fonction avec les droits du créateur (super-admin SQL), permettant de contourner RLS pour des opérations légitimes comme l'inscription.

6.7 — Row Level Security (RLS)

RLS est la couche de sécurité de Supabase. Chaque requête SQL est automatiquement filtrée selon des règles définies pour chaque table. Même si un attaquant connaît l'API, il ne voit que SES données. 

-- Exemple : motos — chaque admin voit seulement les motos de son entreprise
CREATE POLICY "enterprise_members_see_their_motos"
ON motos FOR SELECT
USING (
  -- L'utilisateur appartient à cette entreprise
  enterprise_id IN (
    SELECT enterprise_id FROM enterprise_users
    WHERE user_id = auth.uid() AND is_active = true
    UNION
    SELECT enterprise_id FROM enterprise_admins
    WHERE user_id = auth.uid() AND is_active = true
  )
  OR
  -- Ou c'est le super-admin
  EXISTS (SELECT 1 FROM super_admins WHERE user_id = auth.uid())
);

-- Feed public : toutes les motos disponibles (pour les clients)
CREATE POLICY "public_feed_available_motos"
ON motos FOR SELECT
USING (
  statut = 'disponible'
  AND EXISTS (
    SELECT 1 FROM enterprises e
    WHERE e.id = enterprise_id AND e.is_active = true
  )
);

Chapitre 7Architecture Multi-Tenant

SenMoto sert plusieurs entreprises indépendantes. Chaque entreprise (tenant) a ses propres données, utilisateurs et paramètres, complètement isolés. Voyons comment lib/multitenant.ts orchestre tout cela.

7.1 — Créer une entreprise

// lib/multitenant.ts — Fonction createEnterprise complète annotée

// Génère un code unique type ENT-A4K7P2
function generateEnterpriseCode(): string {
  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
  let code = 'ENT-';
  for (let i = 0; i < 6; i++) {
    code += chars[Math.floor(Math.random() * chars.length)];
  }
  return code;
}

export async function createEnterprise(params: {
  name: string;
  phone?: string;
  email?: string;
  logoUrl?: string;
  adminFullName?: string;
  adminEmail: string;
  adminPhone?: string;
  adminPassword: string;
  latitude?: number;
  longitude?: number;
}) {
  try {
    // Étape 1 : Créer le compte auth de l'admin
    const { data: authData, error: authError } = await supabase.auth.signUp({
      email: params.adminEmail,
      password: params.adminPassword,
    });

    if (authError) throw authError;
    const adminUserId = authData.user!.id;

    // Étape 2 : Générer code unique (retry si doublon)
    let code = '';
    let codeUnique = false;
    while (!codeUnique) {
      code = generateEnterpriseCode();
      const { data } = await supabase
        .from('enterprises')
        .select('id')
        .eq('code', code)
        .single();
      if (!data) codeUnique = true;
    }

    // Étape 3 : Créer entreprise + admin via RPC (transaction atomique)
    const { data: rpcData, error: rpcError } = await supabase.rpc(
      'create_enterprise_with_admin',
      {
        p_name: params.name,
        p_code: code,
        p_phone: params.phone ?? null,
        p_email: params.email ?? null,
        p_logo_url: params.logoUrl ?? null,
        p_admin_user_id: adminUserId,
        p_admin_full_name: params.adminFullName ?? null,
        p_admin_email: params.adminEmail,
        p_admin_phone: params.adminPhone ?? null,
      }
    );
    if (rpcError) throw rpcError;

    // Étape 4 : Sauvegarder la localisation (si fournie)
    if (params.latitude && params.longitude) {
      await supabase.from('entreprise_parametres').upsert({
        entreprise_id: rpcData.enterprise_id,
        localisation: `${params.latitude},${params.longitude}`,
      });
    }

    return { success: true, code };

  } catch (error: any) {
    return { success: false, error: error.message };
  }
}

7.2 — Rejoindre une entreprise

export async function joinEnterprise(params: {
  code: string;        // Code partagé par l'admin
  fullName: string;
  email: string;
  phone?: string;
  password: string;
}) {
  // 1. Vérifier que l'entreprise existe et est active
  const { data: enterprise } = await supabase
    .from('enterprises')
    .select('id, is_active')
    .eq('code', params.code.toUpperCase())
    .single();

  if (!enterprise) return { success: false, error: 'Code invalide' };
  if (!enterprise.is_active) return { success: false, error: 'Entreprise inactive' };

  // 2. Créer le compte utilisateur
  const { data: authData, error: authError } = await supabase.auth.signUp({
    email: params.email,
    password: params.password,
  });
  if (authError) return { success: false, error: authError.message };

  // 3. Ajouter à enterprise_users (is_active=false = attente admin)
  await supabase.from('enterprise_users').insert({
    enterprise_id: enterprise.id,
    user_id: authData.user!.id,
    role: 'user',
    is_active: false,   // Admin doit approuver
  });

  return { success: true };
}

// Créer un user directement depuis l'admin (astuce session)
export async function createUserInEnterprise(params: {
  fullName: string;
  email: string;
  phone?: string;
  password: string;
  enterpriseId: string;
}) {
  // Sauvegarder session admin AVANT signUp
  const { data: { session: adminSession } } = await supabase.auth.getSession();

  // signUp() connecte automatiquement le nouvel utilisateur → perd session admin!
  const { data, error } = await supabase.auth.signUp({
    email: params.email,
    password: params.password,
  });
  if (error) return { success: false, error: error.message };

  // Ajouter à l'entreprise (is_active=true car admin crée directement)
  await supabase.from('enterprise_users').insert({
    enterprise_id: params.enterpriseId,
    user_id: data.user!.id,
    role: 'user',
    is_active: true,
  });

  // Restaurer la session admin
  if (adminSession) {
    await supabase.auth.setSession({
      access_token: adminSession.access_token,
      refresh_token: adminSession.refresh_token,
    });
  }

  return { success: true };
}
⚠️ Piège : signUp() change la session active Quand l'admin crée un utilisateur via signUp(), Supabase connecte automatiquement le nouvel utilisateur et déconnecte l'admin. La solution est de sauvegarder puis restaurer la session admin après la création.

8.2 — Demander les permissions

// lib/notifications.ts
import * as Notifications from 'expo-notifications';
import { Platform } from 'react-native';

export async function ensureNotificationPermissions(): Promise<boolean> {
  const { status: existing } = await Notifications.getPermissionsAsync();

  if (existing === 'granted') return true;

  const { status } = await Notifications.requestPermissionsAsync({
    ios: {
      allowAlert: true,
      allowBadge: true,
      allowSound: true,
    },
  });

  // Android 13+ : permission runtime nécessaire
  if (Platform.OS === 'android' && Platform.Version >= 33) {
    const { requestPermissionsAsync } =
      await import('expo-modules-core');
    // expo-notifications gère cela automatiquement via requestPermissionsAsync()
  }

  return status === 'granted';
}

8.3 — Programmer des rappels

// lib/notifications.ts — Programmer J-3, J-2, Jour-J
export async function scheduleRdvReminders(
  userId: string,
  rdv: RendezVousLite
): Promise<void> {
  if (!rdv.date_rdv) return;

  // Annuler les anciens rappels pour ce RDV
  await cancelRdvReminders(rdv.id);

  const granted = await ensureNotificationPermissions();
  if (!granted) return;

  const dateRdv = new Date(rdv.date_rdv);

  // Définir J-3, J-2, Jour-J
  const rappels = [
    { type: 'rdv_rappel_j3', daysBefore: 3, heure: 9, minute: 0 },
    { type: 'rdv_rappel_j2', daysBefore: 2, heure: 9, minute: 0 },
    { type: 'rdv_rappel_j0', daysBefore: 0, heure: 9, minute: 0 },
  ];

  // Pour Jour-J : utiliser heure du RDV - 1h
  if (rdv.heure_rdv) {
    const [h, m] = rdv.heure_rdv.split(':').map(Number);
    let notifHeure = h - 1;
    let notifMinute = m;
    if (notifHeure < 0) { notifHeure = 9; notifMinute = 0; }
    rappels[2].heure = notifHeure;
    rappels[2].minute = notifMinute;
  }

  for (const rappel of rappels) {
    const scheduledDate = new Date(dateRdv);
    scheduledDate.setDate(scheduledDate.getDate() - rappel.daysBefore);
    scheduledDate.setHours(rappel.heure, rappel.minute, 0, 0);

    // Ne programmer que si la date est dans le futur
    if (scheduledDate <= new Date()) continue;

    // Programmer la notification locale
    const pushId = await Notifications.scheduleNotificationAsync({
      content: {
        title: '📅 Rappel de rendez-vous',
        body: `RDV avec ${rdv.nom_prenom} le ${rdv.date_rdv}`,
        data: { rendezvous_id: rdv.id },
      },
      trigger: {
        type: Notifications.SchedulableTriggerInputTypes.DATE,
        date: scheduledDate,
      },
    });

    // Enregistrer en base pour historique et annulation future
    await supabase.from('notifications').insert({
      user_id: userId,
      type: rappel.type,
      titre: 'Rappel de rendez-vous',
      message: `RDV avec ${rdv.nom_prenom}`,
      rendezvous_id: rdv.id,
      statut: 'pending',
      scheduled_at: scheduledDate.toISOString(),
      push_id: pushId,  // Stocker l'ID pour pouvoir annuler
      lu: false,
    });
  }
}

// Annuler tous les rappels d'un RDV
export async function cancelRdvReminders(rendezvousId: string) {
  // Récupérer les IDs des notifications programmées
  const { data: notifs } = await supabase
    .from('notifications')
    .select('push_id')
    .eq('rendezvous_id', rendezvousId)
    .not('push_id', 'is', null);

  // Annuler chaque notification locale
  if (notifs) {
    for (const n of notifs) {
      await Notifications.cancelScheduledNotificationAsync(n.push_id);
    }
  }

  // Marquer comme annulées en base
  await supabase
    .from('notifications')
    .update({ statut: 'cancelled' })
    .eq('rendezvous_id', rendezvousId);
}

8.4 — Hook : usePushNotifications

// lib/usePushNotifications.ts — À utiliser dans RootLayout
export function usePushNotifications() {
  useEffect(() => {
    let cleanup: (() => void) | undefined;

    const init = async () => {
      // Configurer comment afficher les notifs quand l'app est au premier plan
      Notifications.setNotificationHandler({
        handleNotification: async () => ({
          shouldShowAlert: true,
          shouldPlaySound: true,
          shouldSetBadge: true,
        }),
      });

      // Listener : quand l'utilisateur TAPE sur une notification
      const sub = Notifications.addNotificationResponseReceivedListener(
        (response) => {
          const data = response.notification.request.content.data;
          if (data?.rendezvous_id) {
            // Naviguer vers le RDV concerné
            router.push('/rendezvous');
          }
        }
      );

      cleanup = () => Notifications.removeNotificationSubscription(sub);
    };

    init();
    return () => cleanup?.();
  }, []);
}

}

import { CameraView, useCameraPermissions } from 'expo-camera';

function ScannerMoto() {

      facing="back"
      barcodeScannerSettings={{
        barcodeTypes: ['qr'],  // QR codes seulement
      }}
      onBarcodeScanned={handleBarCodeScanned}
    />
  );
}

9.5 — Images base64 : format et taille

// Le projet stocke les images en base64 dans Supabase
// Format : data:image/jpeg;base64,/9j/4AAQSkZJRgAB...

// lib/storageUtils.ts — Calculer la taille d'une image base64
function estimateBase64Size(base64String: string): number {
  if (!base64String) return 0;
  // Supprimer le préfixe data:image/...;base64,
  const raw = base64String.split(',')[1] ?? base64String;
  // base64 encode 3 bytes en 4 chars → taille réelle ≈ len * 3/4
  return Math.round(raw.length * 3 / 4);
}

// Calculer la taille totale de toutes les images d'une entreprise
export async function calculateTotalStorage(
  userId: string
): Promise<StorageStats> {
  // Récupérer images des motos
  const { data: motoImages } = await supabase
    .from('moto_images')
    .select('image_uri')
    .eq('enterprise_id', enterpriseId);

  const motosSize = motoImages?.reduce(
    (sum, img) => sum + estimateBase64Size(img.image_uri), 0
  ) ?? 0;

  return { total: motosSize, motos: motosSize, /* ... autres */ };
}

10.2 — lib/sharePdf.ts

// lib/sharePdf.ts — Génération et partage PDF
import * as Print from 'expo-print';
import * as Sharing from 'expo-sharing';
import * as FileSystem from 'expo-file-system';
import { Platform } from 'react-native';

export async function printAndSharePdf(
  html: string,
  dialogTitle: string
): Promise<void> {
  try {
    if (Platform.OS === 'ios') {
      // iOS : expo-print peut écrire directement dans le cache
      const { uri } = await Print.printToFileAsync({ html });
      await Sharing.shareAsync(uri, {
        mimeType: 'application/pdf',
        dialogTitle,
        UTI: 'com.adobe.pdf',
      });

    } else {
      // Android : expo-print écrit dans un chemin inaccessible
      // Solution : récupérer le base64 et écrire manuellement
      const { base64 } = await Print.printToFileAsync({
        html,
        base64: true,  // Retourner base64 au lieu d'écrire
      });

      if (!base64) throw new Error('Pas de base64');

      // Écrire dans le cache de l'application
      const fileName = `${dialogTitle.replace(/\s+/g, '-')}-${Date.now()}.pdf`;
      const filePath = `${FileSystem.cacheDirectory}` + fileName;

      await FileSystem.writeAsStringAsync(filePath, base64, {
        encoding: FileSystem.EncodingType.Base64,
      });

      await Sharing.shareAsync(filePath, {
        mimeType: 'application/pdf',
        dialogTitle,
      });
    }
  } catch (error) {
    console.error('Erreur génération PDF:', error);
    Alert.alert('Erreur', 'Impossible de générer le PDF');
  }
}

11.3 — Communication React Native ↔ WebView

// Flux de communication complet

// 1. Utilisateur clique "Ma position" dans la WebView
// 2. WebView envoie message à React Native
window.ReactNativeWebView.postMessage(JSON.stringify({ type: 'LOCATE_ME' }));

// 3. React Native reçoit le message dans onMessage
const handleMessage = async (event) => {
  const raw = JSON.parse(event.nativeEvent.data);

  if (raw.type === 'LOCATE_ME') {
    // 4. React Native demande la permission GPS native
    const { status } = await Location.requestForegroundPermissionsAsync();

    if (status !== 'granted') {
      // 5a. Échec → injecter JS d'erreur dans WebView
      webViewRef.current?.injectJavaScript(
        `locateError("Permission refusée"); true;`
      );
      return;
    }

    // 5b. Succès → obtenir position et l'injecter dans WebView
    const pos = await Location.getCurrentPositionAsync({
      accuracy: Location.Accuracy.High
    });
    const { latitude, longitude } = pos.coords;

    webViewRef.current?.injectJavaScript(
      `receiveLocation(${latitude}, ${longitude}); true;`
    );
    return;
  }

  // 6. Utilisateur confirme position → onLocationSelected
  if (raw.latitude && raw.longitude) {
    onLocationSelected(raw);
    onClose();
  }
};

// Composant WebView dans le rendu
<WebView
  ref={webViewRef}
  source={{ html: buildMapHtml(lat, lng) }}
  javaScriptEnabled
  onMessage={handleMessage}    // Recevoir messages JS → RN
  originWhitelist={['*']}
/>

Chapitre 12Animations

SenMoto utilise les animations pour le menu hamburger et les transitions de pages. Il y a deux approches : l'API Animated native de React Native (simple) et react-native-reanimated (performant, sur le thread natif).

12.1 — Animated API : menu hamburger

// components/HamburgerMenu.tsx — Menu slide-in animé
import { Animated, Dimensions } from 'react-native';

const MENU_WIDTH = Dimensions.get('window').width * 0.80;  // 80% de l'écran

function HamburgerMenu() {
  const [isOpen, setIsOpen] = useState(false);

  // Valeur animée : position X du menu (-MENU_WIDTH quand fermé, 0 quand ouvert)
  const slideAnim = useRef(new Animated.Value(-MENU_WIDTH)).current;

  // Valeur animée : opacité de l'overlay sombre derrière le menu
  const overlayOpacity = useRef(new Animated.Value(0)).current;

  const openMenu = () => {
    setIsOpen(true);
    // Animer les deux valeurs en parallèle
    Animated.parallel([
      Animated.timing(slideAnim, {
        toValue: 0,          // Glisser vers la droite (position 0)
        duration: 280,       // 280ms
        useNativeDriver: true,  // Exécute sur le thread natif (plus fluide)
      }),
      Animated.timing(overlayOpacity, {
        toValue: 0.5,        // Demi-transparent
        duration: 280,
        useNativeDriver: true,
      }),
    ]).start();               // Démarrer l'animation
  };

  const closeMenu = () => {
    Animated.parallel([
      Animated.timing(slideAnim, {
        toValue: -MENU_WIDTH,   // Glisser vers la gauche (hors écran)
        duration: 240,
        useNativeDriver: true,
      }),
      Animated.timing(overlayOpacity, {
        toValue: 0,
        duration: 240,
        useNativeDriver: true,
      }),
    ]).start(() => setIsOpen(false));  // Callback après animation
  };

  return (
    <>
      {/* Contenu principal */}
      {children}

      {/* Overlay sombre (s'affiche quand menu ouvert) */}
      {isOpen && (
        <Animated.View
          style={{ opacity: overlayOpacity, ...StyleSheet.absoluteFill }}
        >
          <TouchableOpacity
            style={{ flex: 1, backgroundColor: 'black' }}
            onPress={closeMenu}
          />
        </Animated.View>
      )}

      {/* Menu lui-même : glisse depuis la gauche */}
      <Animated.View
        style={{
          position: 'absolute',
          top: 0, bottom: 0, left: 0,
          width: MENU_WIDTH,
          transform: [{ translateX: slideAnim }],  // Animation ici
          backgroundColor: '#1E1E1E',
        }}
      >
        {/* Contenu du menu */}
      </Animated.View>
    </>
  );
}

12.2 — react-native-reanimated

// Pour des animations haute performance (60fps garanti)
import Animated, { useSharedValue, useAnimatedStyle, withSpring }
  from 'react-native-reanimated';

function BoutonAnimé() {
  const scale = useSharedValue(1);  // Valeur sur thread natif

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [{ scale: scale.value }],
  }));

  return (
    <Animated.View style={animatedStyle}>
      <TouchableOpacity
        onPressIn={() => { scale.value = withSpring(0.95); }}
        onPressOut={() => { scale.value = withSpring(1); }}
      >
        <Text>Appuyer</Text>
      </TouchableOpacity>
    </Animated.View>
  );
}
